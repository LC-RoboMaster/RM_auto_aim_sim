cmake_minimum_required(VERSION 3.16)
project(rm_auto_aim_sim VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(ENABLE_ROS2 "Build the ROS2 Humble bridge" ON)
option(BUILD_VIEWER "Build the official MuJoCo simulate UI" ON)
option(BUILD_TESTING "Build physics and rendering checks" ON)
set(CONDA_PYTHON "" CACHE FILEPATH "Python executable containing mujoco==3.3.7")
set(YAML_CPP_SOURCE_DIR "" CACHE PATH "Optional existing yaml-cpp source tree")
set(GLFW_INCLUDE_DIR "" CACHE PATH "Optional GLFW headers directory")
set(GLFW_LIBRARY "" CACHE FILEPATH "Optional GLFW shared library")
if(NOT CONDA_PYTHON)
  if(DEFINED ENV{CONDA_PREFIX})
    set(CONDA_PYTHON "$ENV{CONDA_PREFIX}/bin/python")
  else()
    find_program(CONDA_PYTHON_AUTO NAMES python3 python REQUIRED)
    set(CONDA_PYTHON "${CONDA_PYTHON_AUTO}")
  endif()
endif()
execute_process(COMMAND "${CONDA_PYTHON}" -c
  "import mujoco,pathlib; print(mujoco.__version__); print(pathlib.Path(mujoco.__file__).parent)"
  OUTPUT_VARIABLE MUJOCO_INFO OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE MUJOCO_RESULT)
if(NOT MUJOCO_RESULT EQUAL 0)
  message(FATAL_ERROR "Cannot import MuJoCo: set -DCONDA_PYTHON=/path/to/env/bin/python")
endif()
string(REPLACE "\n" ";" MUJOCO_INFO "${MUJOCO_INFO}")
list(GET MUJOCO_INFO 0 MUJOCO_VERSION)
list(GET MUJOCO_INFO 1 MUJOCO_DIR)
if(NOT MUJOCO_VERSION STREQUAL "3.3.7")
  message(FATAL_ERROR "Bundled viewer is pinned to MuJoCo 3.3.7; found ${MUJOCO_VERSION}")
endif()
find_library(MUJOCO_LIBRARY NAMES mujoco libmujoco.so.3.3.7 PATHS "${MUJOCO_DIR}" NO_DEFAULT_PATH REQUIRED)
add_library(mujoco_pip SHARED IMPORTED)
set_target_properties(mujoco_pip PROPERTIES IMPORTED_LOCATION "${MUJOCO_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${MUJOCO_DIR}/include")
if(YAML_CPP_SOURCE_DIR)
  set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${YAML_CPP_SOURCE_DIR}" yaml-cpp)
else()
  find_package(yaml-cpp REQUIRED)
endif()
find_package(Threads REQUIRED)
add_library(aim_core src/config.cpp src/engine.cpp src/projectile_pool.cpp src/target.cpp src/camera_model.cpp src/environment.cpp)
target_include_directories(aim_core PUBLIC include)
target_link_libraries(aim_core PUBLIC mujoco_pip yaml-cpp Threads::Threads)
target_compile_options(aim_core PRIVATE -Wall -Wextra -Wpedantic)
find_package(OpenGL REQUIRED)
add_library(aim_camera src/camera_renderer.cpp src/hdr_pipeline.cpp)
target_link_libraries(aim_camera PUBLIC aim_core OpenGL::GL)

if(BUILD_VIEWER)
  if(GLFW_LIBRARY AND GLFW_INCLUDE_DIR)
    add_library(glfw SHARED IMPORTED)
    set_target_properties(glfw PROPERTIES IMPORTED_LOCATION "${GLFW_LIBRARY}"
      INTERFACE_INCLUDE_DIRECTORIES "${GLFW_INCLUDE_DIR}")
  else()
    find_package(glfw3 3.3 REQUIRED)
  endif()
  set(SIM third_party/simulate)
  add_executable(rm_auto_aim_sim src/main.cpp src/viewer_hooks.cpp src/sensor_worker.cpp
    ${SIM}/simulate.cc ${SIM}/glfw_adapter.cc ${SIM}/glfw_dispatch.cc
    ${SIM}/platform_ui_adapter.cc third_party/lodepng/lodepng.cpp)
  target_compile_definitions(rm_auto_aim_sim PRIVATE GLFW_INCLUDE_NONE)
  target_include_directories(rm_auto_aim_sim PRIVATE ${SIM} third_party/lodepng)
  target_link_libraries(rm_auto_aim_sim PRIVATE aim_camera glfw ${CMAKE_DL_LIBS})
  if(ENABLE_ROS2)
    find_package(ament_cmake REQUIRED)
    foreach(package rclcpp rcl_interfaces sensor_msgs geometry_msgs rosgraph_msgs std_msgs tf2_msgs)
      find_package(${package} REQUIRED)
    endforeach()
    target_sources(rm_auto_aim_sim PRIVATE src/ros_bridge.cpp)
    ament_target_dependencies(rm_auto_aim_sim PUBLIC rclcpp rcl_interfaces sensor_msgs geometry_msgs
      rosgraph_msgs std_msgs tf2_msgs)
  else()
    target_sources(rm_auto_aim_sim PRIVATE src/ros_bridge_stub.cpp)
  endif()
  set_target_properties(rm_auto_aim_sim PROPERTIES BUILD_RPATH "${MUJOCO_DIR}"
    INSTALL_RPATH "${MUJOCO_DIR}")
  install(TARGETS rm_auto_aim_sim DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY config launch scenes docs DESTINATION share/${PROJECT_NAME})
  install(PROGRAMS scripts/auto_control_demo.py scripts/check_ros.py scripts/import_rmuc_field.py
    DESTINATION lib/${PROJECT_NAME})
endif()

if(BUILD_TESTING)
  enable_testing()
  add_executable(aim_core_tests tests/core_tests.cpp)
  target_link_libraries(aim_core_tests PRIVATE aim_core)
  add_test(NAME gimbal_projectile_lifecycle COMMAND aim_core_tests
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_config.yaml")
  add_executable(aim_lab_tests tests/lab_tests.cpp)
  target_link_libraries(aim_lab_tests PRIVATE aim_core)
  add_test(NAME armor_target_statistics COMMAND aim_lab_tests
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_config.yaml")
  add_library(aim_render_probe SHARED tests/render_probe.cpp)
  target_link_libraries(aim_render_probe PRIVATE aim_camera)
endif()
if(ENABLE_ROS2)
  if(NOT BUILD_VIEWER)
    message(FATAL_ERROR "ENABLE_ROS2 requires BUILD_VIEWER=ON")
  endif()
  ament_package()
endif()
